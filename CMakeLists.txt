option(LOINA_DIFFUSION "Build the image adapter" ON)
option(LOINA_SPEECH "Build the Parakeet speech adapter" ON)

option(LOINA_TESTS "Build model-free regression tests" ON)

option(LOINA_EVAL "Build the fixed-evidence chat evaluation runner" OFF)
set(LOINA_RUNTIME_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/.." CACHE PATH "Directory containing Loina runtime forks")

set(LOINA_SOURCES loina.c loina_util.c loina_device.c loina_package.c
    loina_catalog.c loina_session.c loina_cache.c loina_chat.c loina_web.c loina_media.c loina_templates.c)
file(GLOB template_files CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/templates/*.jinja" "${CMAKE_CURRENT_SOURCE_DIR}/templates/*.json")
set(template_header "static const struct { const char *name, *source; } loina_templates[] = {\n")
foreach(template_file IN LISTS template_files)
    set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${template_file}")
    get_filename_component(template_name "${template_file}" NAME)
    file(READ "${template_file}" template_source HEX)
    string(REGEX REPLACE "(..)" "0x\\1," template_source "${template_source}")
    string(APPEND template_header "{ \"${template_name}\", (const char[]) { ${template_source}0 } },\n")
endforeach()
string(APPEND template_header "};\n")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/loina_templates.h" "${template_header}")
add_library(loina-core STATIC ${LOINA_SOURCES})
target_include_directories(loina-core PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
target_include_directories(loina-core PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
if(NOT MSVC)
    target_compile_options(loina-core PRIVATE -Wall -Wextra -Wpedantic)
endif()
if(NOT WIN32)
    target_compile_definitions(loina-core PRIVATE _POSIX_C_SOURCE=200809L _DEFAULT_SOURCE)
    target_link_libraries(loina-core PUBLIC m)
endif()

    set(GGML_OPENMP OFF CACHE BOOL "" FORCE)

    set(LLAMA_BUILD_COMMON ${LOINA_BENCH} CACHE BOOL "" FORCE)
    set(LLAMA_BUILD_JINJA ${LOINA_LLAMA} CACHE BOOL "" FORCE)
    set(LLAMA_BUILD_TESTS OFF CACHE BOOL "" FORCE)
    set(LLAMA_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)

        add_subdirectory("${LOINA_RUNTIME_ROOT}/llama.cpp/tools/llama-bench" runtimes/llama-bench)
    endif()
endif()

function(loina_revision directory macro)
    execute_process(COMMAND git -C "${directory}" rev-parse HEAD
        OUTPUT_VARIABLE revision OUTPUT_STRIP_TRAILING_WHITESPACE ERROR_QUIET)
    if(NOT revision)
        message(FATAL_ERROR "Runtime ${directory} needs a git revision for cache compatibility")
    endif()
    target_compile_definitions(loina-core PRIVATE "${macro}=\"${revision}\"")
endfunction()
